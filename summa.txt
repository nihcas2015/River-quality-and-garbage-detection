cd ~/pi4_edge_node

# ── 1. Clean slate ──
rm -rf venv
sudo apt remove -y python3-simplejpeg

# ── 2. Install ALL system packages (these are compiled together, guaranteed compatible) ──
sudo apt update
sudo apt install -y python3-picamera2 python3-numpy python3-opencv \
                    python3-pil python3-requests python3-matplotlib \
                    python3-pandas python3-scipy python3-tqdm python3-yaml \
                    python3-psutil

# ── 3. Create venv that sees system packages ──
python3 -m venv --system-site-packages venv
source venv/bin/activate

# ── 4. Install ONLY what the system doesn't provide ──
pip install paho-mqtt==1.6.1

# ── 5. Install ultralytics WITHOUT pulling numpy/opencv/etc ──
pip install ultralytics --no-deps

# ── 6. Install PyTorch (the one dep system doesn't have) ──
pip install torch torchvision --extra-index-url https://download.pytorch.org/whl/cpu

# ── 7. Safety cleanup — remove anything pip snuck in that shadows system packages ──
pip uninstall -y numpy opencv-python opencv-python-headless 2>/dev/null; true

# ── 8. Verify everything ──
python3 -c "import numpy; print('numpy', numpy.__version__, numpy.__file__)"
python3 -c "import cv2; print('cv2', cv2.__version__)"
python3 -c "from picamera2 import Picamera2; print('picamera2 OK')"
python3 -c "from ultralytics import YOLO; print('YOLO OK')"
python3 -c "import paho.mqtt; print('mqtt OK')"
